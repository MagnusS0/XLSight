void Measure(string name, Func<ExcelWorkbook, long> op)
{
    GC.Collect(); GC.WaitForPendingFinalizers(); GC.Collect();
    long before = GC.GetTotalMemory(true);
    long a0 = GC.GetTotalAllocatedBytes(true);
    var wb = ExcelWorkbook.Open(f);
    long result = op(wb);
    long alloc = GC.GetTotalAllocatedBytes(true) - a0;
    long retainedOpen = GC.GetTotalMemory(true) - before;
    wb.Dispose();
    long retainedDisposed = GC.GetTotalMemory(true) - before;
    GC.KeepAlive(wb);
    Console.WriteLine($"{name,-9} result={result} alloc={alloc / 1024} KiB retained(open)={retainedOpen / 1024} KiB retained(disposed, ref held)={retainedDisposed / 1024} KiB");
}
